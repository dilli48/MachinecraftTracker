DROP TABLE IF EXISTS components;

CREATE TABLE components (
    id SERIAL PRIMARY KEY,
    part_number VARCHAR(100) NOT NULL UNIQUE,
    type VARCHAR(100),
    footprint VARCHAR(100),
    current_stock INTEGER NOT NULL DEFAULT 0,
    minimum_threshold INTEGER NOT NULL DEFAULT 0,
    comments TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

Launch Uvicorn server (./venv/bin/uvicorn main:app --host 127.0.0.1 --port 8000).
Use the browser subagent tool to open http://localhost:8000/.

curl -I http://127.0.0.1:8000/ && curl -I http://127.0.0.1:8000/operator && curl -s http://127.0.0.1:8000/static/operator/manifest.json
